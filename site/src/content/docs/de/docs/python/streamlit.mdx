```bash

```

```python
import streamlit as st
import streamlit.components.v1 as components
from signavis import render_daw_player

f = st.file_uploader("Aufnahme", type=["wav", "mp3", "ogg", "flac"])
if f:
    components.html(render_daw_player(f.read(), viewMode="spectrogram"), height=640)
```
